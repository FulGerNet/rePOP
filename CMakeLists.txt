cmake_minimum_required(VERSION 3.22)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(reWW LANGUAGES C CXX RC)
# A failed initial MSVC discovery can leave empty Release flags cached.
# Restore the normal MSVC defaults without changing an existing custom set.
if(MSVC)
  foreach(language C CXX)
    if("${CMAKE_${language}_FLAGS_RELEASE}" STREQUAL "")
      set(CMAKE_${language}_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG" CACHE STRING "Release compiler flags" FORCE)
    endif()
  endforeach()
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "reWW requires x64")
endif()
option(REWW_ABI_CHECK "Validate guest callee-saved registers" OFF)
if(REWW_ABI_CHECK)
  add_compile_definitions(RECOMP_ABI_CHECK)
endif()
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX REWW_RUNTIME=1 RESOT_RUNTIME_RETAIL=1)
if(MSVC)
  set(reww_release_flags "${CMAKE_C_FLAGS_RELEASE}")
  set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} /Zi")
endif()
add_subdirectory(port/third_party/xboxrecomp)
option(REWW_BUILD_TESTS "Build loader integration checks" ON)
if(REWW_BUILD_TESTS)
  add_subdirectory(tests)
endif()
if(MSVC)
  set(CMAKE_C_FLAGS_RELEASE "${reww_release_flags}")
endif()
set(target_dir "${CMAKE_CURRENT_SOURCE_DIR}/source/xbox-retail")
set(xbe "${CMAKE_CURRENT_SOURCE_DIR}/Assets/xbox-retail/default.xbe")
file(READ "${target_dir}/target.h" facts)
string(REGEX MATCH "RESOT_XBE_SHA256[ \t]+\"([A-Fa-f0-9]+)\"" match "${facts}")
set(expected "${CMAKE_MATCH_1}")
file(SHA256 "${xbe}" actual)
string(TOUPPER "${actual}" actual)
if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Unsupported WW XBE SHA-256")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${xbe}" "${target_dir}/target.h")
file(GLOB_RECURSE gen CONFIGURE_DEPENDS "${target_dir}/gen/*.c")
if(NOT gen)
  message(FATAL_ERROR "Run tools/recomp.ps1 first")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/guest.rc" CONTENT "101 RCDATA \"${xbe}\"\n")
add_executable(ww-xbox-retail WIN32 port/windows/src/main.c
  port/windows/src/launcher.c port/windows/src/options.c
  port/windows/src/diag.c port/windows/src/scene.c
  port/windows/res/reww.rc port/third_party/extract-xiso/extract-xiso.c
  port/game/recomp_manual.c ${gen} "${CMAKE_CURRENT_BINARY_DIR}/guest.rc")
set_source_files_properties(port/third_party/extract-xiso/extract-xiso.c PROPERTIES
  COMPILE_DEFINITIONS "WIN32;main=resot_extract_xiso_main;_CRT_SECURE_NO_WARNINGS;_CRT_NONSTDC_NO_WARNINGS"
  COMPILE_OPTIONS "$<$<C_COMPILER_ID:MSVC>:/w>")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/port/windows/res/AtlantisMF.ttf")
  target_compile_definitions(ww-xbox-retail PRIVATE RESOT_HAVE_LAUNCHER_FONT=1)
endif()
if(EXISTS "${target_dir}/target_hooks.c")
  target_sources(ww-xbox-retail PRIVATE "${target_dir}/target_hooks.c")
endif()
target_include_directories(ww-xbox-retail PRIVATE "${target_dir}" "${target_dir}/gen" port/third_party/xboxrecomp/include)
# RE-WW (from reSOT): picture shape when RECOMP_HW_ASPECT is not set. "16:9"
# widens Hor+ through target_hooks.c; empty keeps the original 4:3.
set(REWW_ASPECT "16:9" CACHE STRING "Default picture shape (w:h), empty for 4:3")
if(REWW_ASPECT)
  target_compile_definitions(ww-xbox-retail PRIVATE "RESOT_DEFAULT_ASPECT=\"${REWW_ASPECT}\"")
endif()
target_link_libraries(ww-xbox-retail PRIVATE xboxrecomp d3d11 dxgi dxguid xinput winmm dbghelp bcrypt ole32 shell32 uuid msimg32 windowscodecs gdi32 psapi)
set_target_properties(ww-xbox-retail PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(REWW_ABI_CHECK)
  set_target_properties(ww-xbox-retail PROPERTIES OUTPUT_NAME "ww-xbox-retail-abi")
endif()
if(MSVC)
  target_compile_options(ww-xbox-retail PRIVATE /bigobj /MP4)
  # Lifted C frames are larger than the original x86 frames. The default
  # The initial archive-loading overflow reproduced even with 16 MB and was
  # traced to a missing guest cold branch (docs/COLD-BRANCH-RECOVERY.md).
  # Keep this reserve as native frame headroom, separately from that fix.
  target_link_options(ww-xbox-retail PRIVATE /LARGEADDRESSAWARE /DEBUG /STACK:16777216)
endif()
